   600→/** Price formatting per language. */
   601→export function priceFmt(lang: Lang, n: number, currency: "MAD" | "EUR" = "MAD"): string {
   602→  const localeMap: Record<Lang, string> = { en: "en-GB", fr: "fr-FR", de: "de-DE", es: "es-ES", nl: "nl-NL" };
   603→  const suffix = currency === "MAD" ? "MAD" : "EUR";
   604→  return `${n.toLocaleString(localeMap[lang])} ${suffix}`;
   605→}
